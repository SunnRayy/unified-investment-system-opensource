# src/database/mapping_seeds.py mixes generic/structural definitions with the
# owner's real Financial-Summary column names and historical figures. It
# cannot simply be dropped (src.database.connector's migrations import it
# unconditionally on any fresh --init; ~8 other modules import it at module
# level or via routine lazy imports that bypass UIS_SEED_PROFILE entirely —
# see the WS-7 report). export_public.sh instead COPIES OVER it with
# tools/release/mapping_seeds.public.py, the persona-safe structural twin.
EXCLUDE src/database/mapping_seeds.py

# Real-data analysis script, not part of the shipped product surface.
EXCLUDE scripts/analyze_template_columns.py

# Tests scripts/backfill_rule_buckets.py, which isn't on the scripts/
# keeper list above — importing it would ImportError in a fresh clone.
EXCLUDE tests/scripts/test_backfill_rule_buckets.py

# Contains the exact strings it forbids — publishing it would defeat the gate.
EXCLUDE tools/release/leak_patterns.txt

# Same reason as leak_patterns.txt: a triaged-safe allowlist necessarily
# quotes the real strings/figures it's excusing (fund codes, docstring

# tracks are already resolved in what actually ships. Found 2026-08-17 —
# leak_gate.py's own SELF_EXEMPT only matches by absolute path, so it
# doesn't cover the staged COPY of accepted_findings.txt either (same gap
# already worked around for leak_gate.py's own docstring citations).
